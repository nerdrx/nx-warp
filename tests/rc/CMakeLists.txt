if(NOT TARGET nxvc_rc)
  return()
endif()

foreach(t classify foveation allocate model governor tvm refresh)
  add_executable(rc_test_${t} test_${t}.cpp)
  target_link_libraries(rc_test_${t} PRIVATE nxvc_rc)
  target_include_directories(rc_test_${t} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  set_target_properties(rc_test_${t} PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON)
  if(NOT MSVC)
    target_compile_options(rc_test_${t} PRIVATE -Wall -Wextra -Wno-unused-parameter)
  endif()
  add_test(NAME rc.${t} COMMAND rc_test_${t})
endforeach()
