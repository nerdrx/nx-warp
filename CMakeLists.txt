cmake_minimum_required(VERSION 3.25)
project(nxwarp VERSION 0.0.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(NXWARP_BUILD_REF "Build the CPU reference codec" ON)
option(NXWARP_BUILD_VK "Build the Vulkan encoder/decoder" OFF)
option(NXWARP_BUILD_TOOLS "Build tools" ON)
option(NXWARP_BUILD_TESTS "Build tests" ON)
if(NXWARP_BUILD_REF AND EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/ref/CMakeLists.txt)
  add_subdirectory(ref)
endif()
if(NXWARP_BUILD_VK AND EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/vk/CMakeLists.txt)
  add_subdirectory(vk)
endif()
if(NXWARP_BUILD_TOOLS AND EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/tools/CMakeLists.txt)
  add_subdirectory(tools)
endif()
if(NXWARP_BUILD_TESTS AND EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/tests/CMakeLists.txt)
  enable_testing()
  add_subdirectory(tests)
endif()
