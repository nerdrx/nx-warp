cmake_minimum_required(VERSION 3.22)
project(zstd_dctx_probe LANGUAGES CXX)
if(NOT DEFINED ZSTD_INCLUDE OR NOT DEFINED ZSTD_LIBRARY)
  message(FATAL_ERROR "Pass ZSTD_INCLUDE and ZSTD_LIBRARY")
endif()
add_executable(zstd-dctx-probe src/main.cpp)
target_compile_features(zstd-dctx-probe PRIVATE cxx_std_17)
target_compile_options(zstd-dctx-probe PRIVATE -O3 -DNDEBUG -Wall -Wextra)
target_include_directories(zstd-dctx-probe PRIVATE "${ZSTD_INCLUDE}")
target_link_libraries(zstd-dctx-probe PRIVATE "${ZSTD_LIBRARY}")
