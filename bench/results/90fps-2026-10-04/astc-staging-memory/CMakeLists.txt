cmake_minimum_required(VERSION 3.22)
project(astc_pico LANGUAGES C CXX)
if(NOT DEFINED LZ4_DIR OR NOT EXISTS "${LZ4_DIR}/lz4.c" OR NOT EXISTS "${LZ4_DIR}/lz4.h")
    message(FATAL_ERROR "Pass -DLZ4_DIR=/path/to/LZ4/lib (v1.10.0 tested)")
endif()
if(NOT DEFINED VMA_INCLUDE OR NOT EXISTS "${VMA_INCLUDE}/vk_mem_alloc.h")
    message(FATAL_ERROR "Pass -DVMA_INCLUDE=/path/to/vk_mem_alloc.h directory")
endif()
if(NOT DEFINED ZSTD_INCLUDE OR NOT EXISTS "${ZSTD_INCLUDE}/zstd.h" OR NOT DEFINED ZSTD_LIBRARY OR NOT EXISTS "${ZSTD_LIBRARY}")
    message(FATAL_ERROR "Pass Android -DZSTD_INCLUDE and -DZSTD_LIBRARY")
endif()
add_executable(astc-pico src/main.cpp "${LZ4_DIR}/lz4.c")
target_compile_features(astc-pico PRIVATE c_std_99 cxx_std_17)
target_compile_options(astc-pico PRIVATE -O3 -Wall -Wextra -Wno-missing-field-initializers -Wno-nullability-completeness -Wno-unused-parameter -Wno-unused-private-field)
target_compile_definitions(astc-pico PRIVATE VMA_VULKAN_VERSION=1001000)
target_include_directories(astc-pico PRIVATE "${VULKAN_HEADERS}/include" "${CMAKE_BINARY_DIR}/generated" "${LZ4_DIR}" "${VMA_INCLUDE}" "${ZSTD_INCLUDE}")
target_link_libraries(astc-pico PRIVATE vulkan android log "${ZSTD_LIBRARY}")
