cmake_minimum_required(VERSION 3.20)
project(nxastc_zstd_compact_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_library(ZSTD_LIB zstd REQUIRED)
find_library(LZ4_LIB lz4 REQUIRED)
add_executable(zstd_compact src/main.cpp)
target_include_directories(zstd_compact PRIVATE include)
target_link_libraries(zstd_compact PRIVATE ${ZSTD_LIB} ${LZ4_LIB})
target_compile_options(zstd_compact PRIVATE -O2 -DNDEBUG -Wall -Wextra -Wpedantic)
