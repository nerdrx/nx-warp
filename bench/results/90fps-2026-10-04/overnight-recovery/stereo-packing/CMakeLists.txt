cmake_minimum_required(VERSION 3.20)
project(nx_stereo_pack_bench LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_library(LZ4_LIB lz4 REQUIRED)
find_library(ZSTD_LIB zstd REQUIRED)
add_executable(stereo-pack src/main.cpp)
target_include_directories(stereo-pack PRIVATE include)
target_link_libraries(stereo-pack PRIVATE Threads::Threads ${LZ4_LIB} ${ZSTD_LIB})
target_compile_options(stereo-pack PRIVATE -O3 -Wall -Wextra -Wpedantic)
