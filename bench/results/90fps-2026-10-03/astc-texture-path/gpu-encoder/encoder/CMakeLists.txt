cmake_minimum_required(VERSION 3.20)
project(nx_astc_gpu LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Vulkan REQUIRED)
set(LZ4_DIR "" CACHE PATH "Directory containing lz4.c and lz4.h")
if(NOT EXISTS "${LZ4_DIR}/lz4.c" OR NOT EXISTS "${LZ4_DIR}/lz4.h")
  message(FATAL_ERROR "Set LZ4_DIR to a directory containing lz4.c and lz4.h")
endif()
add_executable(astc-gpu src/main.cpp "${LZ4_DIR}/lz4.c")
target_include_directories(astc-gpu PRIVATE "${LZ4_DIR}")
target_link_libraries(astc-gpu PRIVATE Vulkan::Vulkan)
target_compile_options(astc-gpu PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-O3;-Wall;-Wextra;-Wpedantic>)
